// JavaScript Example
async function getNavigationUrl(source, destination) {
    const response = await fetch('http://localhost:8000/api/navigate', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({ source, destination })
    });
    
    const data = await response.json();
    
    // Use the full URL in your iframe
    document.getElementById('myIframe').src = data.floor_html_url;
    
    console.log('Navigation URL:', data.floor_html_url);
    console.log('Cross-floor:', data.cross_floor);
}

// Usage
getNavigationUrl('Transformation', 'Innovation');
